# Client e Server Side

- Client: tudo que o usuário precisa interagir (botao...)
- Server: tudo que precisa somente renderizar (titulo, label...)

## Limpar `global.css`
```css
@import "tailwindcss";
```

## Limpar `layout.tsx`
```tsx
import type { Metadata } from "next";
import "./globals.css";

export const metadata: Metadata = {
  title: "Create Next App",
  description: "Generated by create next app",
};

export default function RootLayout({ children }: LayoutProps<"/">) {
  return (
    <html lang="en">
      <body>
        {children}
      </body>
    </html>
  );
}
```

## Limpar `page.tsx`
```tsx
export default function HomePage() {
  return <div>Teste</div>
}
```

## Use client
- Utilizar o `use client` para dizer que vai ser renderizado do lado do servidor

```tsx
"use client";

export default function HomePage() {

  function quandoMeClica() {
    alert("foi clicado");
  }

  return (
    <div>
      <h1>Título da página</h1>
      <button onClick={quandoMeClica} >Click-me!</button>
    </div>
  )
}
```

- Recomendado é não user o use client em uma página
- Usar use client nos componentes


### `button.tsx`
```tsx
"use client";

export default function Button() {

    function quandoMeClica() {
        alert("foi clicado");
    }

    return <button onClick={quandoMeClica} >Click-me!</button>
}
```

### `page.tsx`
```tsx
import Button from "./button"

export default function HomePage() {
  return (
    <div>
      <h1>Título da página</h1>
      <Button></Button>
    </div>
  )
}
```

## Definir o que acontece quando clica em submit do botão 
```tsx
"use client";

export default function CreateAccountForm() {
  return (
    <form
      className="flex flex-col gap-2"
      onSubmit={(event) => {
        event.preventDefault(); // impede que a pagina seja recarregada
        alert("Enviei")
      }}>
      <input type="text" placeholder="Digite o seu nome" />
      <input type="text" placeholder="Digite o seu sobrenome" />
      <input type="email" placeholder="Digite o seu email" />
      <input type="password" placeholder="Digite a sua senha" />
      <button type="submit">Enviar</button>
    </form>
  )
}
```

# Hooks
- Necessarios para re-renderizar a pagina

## useState
```ts
const [state, setState] = useState(initialValue);
```

- Exemplo abaixo

`create-account-form.tsx`

```tsx
"use client";

import { useState } from "react";

export default function CreateAccountForm() {

  const [nome, setNome] = useState("Mateus"); // define o nome

  return (
    <form
      className="flex flex-col gap-2"
      onSubmit={(event) => {
        event.preventDefault(); // impede que a pagina seja recarregada
        setNome("Teste") // atualiza o nome
        alert(nome)
      }}>
      <input type="text" placeholder="Digite o seu nome" value={nome} />
      <input type="text" placeholder="Digite o seu sobrenome" />
      <input type="email" placeholder="Digite o seu email" />
      <input type="password" placeholder="Digite a sua senha" />
      <button type="submit">Enviar</button>
    </form>
  )
}
```

## Binding

- A funcao recebe um evento, informando o que aconteceu
- Em `event.target.value`, temos o vamos mais atualizado do campo value
- Apõs isso, o componente é re-render

```tsx
onChange={(event) => setNome(event.target.value)}
```

`create-account-form.tsx`

```tsx
"use client";

import { useState } from "react";

export default function CreateAccountForm() {

  const [nome, setNome] = useState("Mateus"); // define o nome

  return (
    <form
      className="flex flex-col gap-2"
      onSubmit={(event) => {
        event.preventDefault(); // impede que a pagina seja recarregada
        setNome("Teste") // atualiza o nome
      }}>
      <input type="text" placeholder="Digite o seu nome" value={nome} onChange={(event) => setNome(event.target.value)}/>
      <input type="text" placeholder="Digite o seu sobrenome" />
      <input type="email" placeholder="Digite o seu email" />
      <input type="password" placeholder="Digite a sua senha" />
      <button type="submit">Enviar</button>
    </form>
  )
}
```

## useEffect
- Ação que acontece por causa de uma alteração de status
- Permite disparar um evento, quando acontece alguma mudança na aplicação

```tsx
  useEffect(() => {
    alert("Algo mudou");
  }, [nome]);
```
- Toda vez que mudarmos o nome, o useEffect vai ser disparado

`create-account-form.tsx`

```tsx
"use client";

import { useState, useEffect } from "react";

export default function CreateAccountForm() {

  const [nome, setNome] = useState("Mateus"); // define o nome

  useEffect(() => {
  }, [nome]);

  return (
    <form
      className="flex flex-col gap-2"
      onSubmit={(event) => {
        event.preventDefault(); // impede que a pagina seja recarregada
        setNome("Teste") // atualiza o nome
        alert(nome)
      }}>
      <input type="text" placeholder="Digite o seu nome" value={nome} onChange={(event) => setNome(event.target.value)}/>
      <input type="text" placeholder="Digite o seu sobrenome" />
      <input type="email" placeholder="Digite o seu email" />
      <input type="password" placeholder="Digite a sua senha" />
      <button type="submit">Enviar</button>
    </form>
  )
}
```

- Podemos usar o useEffect para atualizar as infos em tempo real

```tsx
"use client";

import { useState, useEffect } from "react";

export default function CreateAccountForm() {

  const [nome, setNome] = useState(""); // define o nome
  const [sobrenome, setSobrenome] = useState("");
  const [email, setEmail] = useState("");
  const [senha, setSenha] = useState("");

  useEffect(() => {
    console.log(`
      Nome: ${nome},
      Sobrenome: ${sobrenome},
      Email: ${email},
      Senha: ${senha},
      `);
  }, [nome, sobrenome, email, senha]);

  return (
    <form
      className="flex flex-col gap-2"
      onSubmit={(event) => {
        event.preventDefault(); // impede que a pagina seja recarregada
        setNome("Teste") // atualiza o nome
        alert(nome)
      }}>
      <input type="text" placeholder="Digite o seu nome" value={nome} onChange={(event) => setNome(event.target.value)}/>
      <input type="text" placeholder="Digite o seu sobrenome" value={sobrenome} onChange={(event) => setSobrenome(event.target.value)} />
      <input type="email" placeholder="Digite o seu email" value={email} onChange={(event) => setEmail(event.target.value)} />
      <input type="password" placeholder="Digite a sua senha" value={senha} onChange={(event) => setSenha(event.target.value)} />
      <button type="submit">Enviar</button>
    </form>
  )
}
```
